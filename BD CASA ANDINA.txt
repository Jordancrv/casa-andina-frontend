/* ============================================================================
   2. TABLAS MAESTRAS / CATÁLOGOS
   ============================================================================ */

-- Sedes: dato maestro fijo. Sin CRUD en frontend (Restricción #2 del Business
-- Case). Se administra únicamente vía scripts SQL por el Responsable de Datos.
CREATE TABLE dbo.Sede (
    SedeId          INT             IDENTITY(1,1)   NOT NULL,
    Nombre          NVARCHAR(150)   NOT NULL,
    Codigo          NVARCHAR(20)    NOT NULL,           -- ej. 'CUS', 'MIR', 'ARQ'
    Region          NVARCHAR(30)    NOT NULL,           -- Costa / Sierra / Selva (dashboard BI)
    Ciudad          NVARCHAR(100)   NOT NULL,
    Direccion       NVARCHAR(250)   NULL,
    Categoria       NVARCHAR(30)    NOT NULL,           -- Premium / Select / Standard
    Telefono        NVARCHAR(20)    NULL,
    Activo          BIT             NOT NULL DEFAULT (1),
    FechaCreacion   DATETIME2       NOT NULL DEFAULT (SYSDATETIME()),
    CONSTRAINT PK_Sede PRIMARY KEY (SedeId),
    CONSTRAINT UQ_Sede_Codigo UNIQUE (Codigo),
    CONSTRAINT CK_Sede_Region CHECK (Region IN (N'Costa', N'Sierra', N'Selva')),
    CONSTRAINT CK_Sede_Categoria CHECK (Categoria IN (N'Premium', N'Select', N'Standard'))
);
GO

-- Roles del personal interno (Hito 3: Administrador, Recepción, Operaciones, Mantenimiento)
CREATE TABLE dbo.Rol (
    RolId           INT             IDENTITY(1,1)   NOT NULL,
    Nombre          NVARCHAR(50)    NOT NULL,
    Descripcion     NVARCHAR(200)   NULL,
    CONSTRAINT PK_Rol PRIMARY KEY (RolId),
    CONSTRAINT UQ_Rol_Nombre UNIQUE (Nombre)
);
GO

-- Tipos de habitación (mockup Calendario: Superior King, Superior Twin, Andina Suite...)
CREATE TABLE dbo.TipoHabitacion (
    TipoHabitacionId    INT             IDENTITY(1,1)   NOT NULL,
    Nombre              NVARCHAR(60)    NOT NULL,
    Descripcion         NVARCHAR(250)   NULL,
    CapacidadAdultos    TINYINT         NOT NULL DEFAULT (2),
    CapacidadNinos      TINYINT         NOT NULL DEFAULT (0),
    CONSTRAINT PK_TipoHabitacion PRIMARY KEY (TipoHabitacionId),
    CONSTRAINT UQ_TipoHabitacion_Nombre UNIQUE (Nombre)
);
GO

-- Comodidades (mockup Nueva Habitación: WiFi, AC, TV/Cable, Jacuzzi)
CREATE TABLE dbo.Comodidad (
    ComodidadId     INT             IDENTITY(1,1)   NOT NULL,
    Nombre          NVARCHAR(60)    NOT NULL,
    CONSTRAINT PK_Comodidad PRIMARY KEY (ComodidadId),
    CONSTRAINT UQ_Comodidad_Nombre UNIQUE (Nombre)
);
GO

/* ============================================================================
   3. HABITACIONES Y SU RELACIÓN N:M CON COMODIDADES
   ============================================================================ */

CREATE TABLE dbo.Habitacion (
    HabitacionId        INT             IDENTITY(1,1)   NOT NULL,
    SedeId               INT             NOT NULL,
    TipoHabitacionId     INT             NOT NULL,
    Numero               NVARCHAR(10)    NOT NULL,          -- ej. '101', '202'
    Piso                 TINYINT         NOT NULL,
    PrecioBase           DECIMAL(10,2)   NOT NULL,
    Estado               NVARCHAR(20)    NOT NULL DEFAULT (N'Disponible'),
    FotoUrl               NVARCHAR(500)   NULL,
    Activo                BIT             NOT NULL DEFAULT (1),
    FechaCreacion         DATETIME2       NOT NULL DEFAULT (SYSDATETIME()),
    CONSTRAINT PK_Habitacion PRIMARY KEY (HabitacionId),
    CONSTRAINT UQ_Habitacion_Sede_Numero UNIQUE (SedeId, Numero),
    CONSTRAINT FK_Habitacion_Sede FOREIGN KEY (SedeId)
        REFERENCES dbo.Sede (SedeId),
    CONSTRAINT FK_Habitacion_TipoHabitacion FOREIGN KEY (TipoHabitacionId)
        REFERENCES dbo.TipoHabitacion (TipoHabitacionId),
    CONSTRAINT CK_Habitacion_Estado CHECK (Estado IN
        (N'Disponible', N'Ocupada', N'Mantenimiento', N'Bloqueada')),
    CONSTRAINT CK_Habitacion_PrecioBase CHECK (PrecioBase >= 0)
);
GO

-- Tabla puente: una habitación puede tener varias comodidades y una
-- comodidad puede estar en varias habitaciones (mockup Nueva Habitación)
CREATE TABLE dbo.HabitacionComodidad (
    HabitacionId    INT     NOT NULL,
    ComodidadId     INT     NOT NULL,
    CONSTRAINT PK_HabitacionComodidad PRIMARY KEY (HabitacionId, ComodidadId),
    CONSTRAINT FK_HabComodidad_Habitacion FOREIGN KEY (HabitacionId)
        REFERENCES dbo.Habitacion (HabitacionId) ON DELETE CASCADE,
    CONSTRAINT FK_HabComodidad_Comodidad FOREIGN KEY (ComodidadId)
        REFERENCES dbo.Comodidad (ComodidadId)
);
GO

/* ============================================================================
   4. SERVICIOS Y SU RELACIÓN N:M CON SEDES
   ============================================================================ */

CREATE TABLE dbo.Servicio (
    ServicioId      INT             IDENTITY(1,1)   NOT NULL,
    Nombre          NVARCHAR(100)   NOT NULL,
    Descripcion     NVARCHAR(300)   NULL,
    Precio          DECIMAL(10,2)   NOT NULL DEFAULT (0),
    Activo          BIT             NOT NULL DEFAULT (1),
    FechaCreacion   DATETIME2       NOT NULL DEFAULT (SYSDATETIME()),
    CONSTRAINT PK_Servicio PRIMARY KEY (ServicioId),
    CONSTRAINT CK_Servicio_Precio CHECK (Precio >= 0)
);
GO

-- Tabla puente: "vinculación múltiple a sedes específicas" (Business Case, 6.1.d)
CREATE TABLE dbo.ServicioSede (
    ServicioId      INT     NOT NULL,
    SedeId          INT     NOT NULL,
    CONSTRAINT PK_ServicioSede PRIMARY KEY (ServicioId, SedeId),
    CONSTRAINT FK_ServicioSede_Servicio FOREIGN KEY (ServicioId)
        REFERENCES dbo.Servicio (ServicioId) ON DELETE CASCADE,
    CONSTRAINT FK_ServicioSede_Sede FOREIGN KEY (SedeId)
        REFERENCES dbo.Sede (SedeId)
);
GO

/* ============================================================================
   5. USUARIOS INTERNOS (Portal Admin) Y CLIENTES (Portal Cliente)
   ============================================================================ */

-- Personal interno: Administrador, Recepción, Operaciones, Mantenimiento
CREATE TABLE dbo.Usuario (
    UsuarioId           INT             IDENTITY(1,1)   NOT NULL,
    RolId                INT             NOT NULL,
    SedeId               INT             NULL,               -- NULL = HQ / acceso multi-sede
    Nombre               NVARCHAR(150)   NOT NULL,
    Correo               NVARCHAR(150)   NOT NULL,
    ContrasenaHash       NVARCHAR(256)   NOT NULL,
    Telefono             NVARCHAR(20)    NULL,
    Estado               NVARCHAR(20)    NOT NULL DEFAULT (N'Activo'),
    UltimoAcceso         DATETIME2       NULL,
    FechaRegistro        DATETIME2       NOT NULL DEFAULT (SYSDATETIME()),
    CONSTRAINT PK_Usuario PRIMARY KEY (UsuarioId),
    CONSTRAINT UQ_Usuario_Correo UNIQUE (Correo),
    CONSTRAINT FK_Usuario_Rol FOREIGN KEY (RolId)
        REFERENCES dbo.Rol (RolId),
    CONSTRAINT FK_Usuario_Sede FOREIGN KEY (SedeId)
        REFERENCES dbo.Sede (SedeId),
    CONSTRAINT CK_Usuario_Estado CHECK (Estado IN (N'Activo', N'Inactivo'))
);
GO

-- Huéspedes / clientes del Portal Cliente
CREATE TABLE dbo.Cliente (
    ClienteId           INT             IDENTITY(1,1)   NOT NULL,
    NombreCompleto       NVARCHAR(200)   NOT NULL,
    DNI                  NVARCHAR(15)    NOT NULL,
    Correo               NVARCHAR(150)   NOT NULL,
    Telefono             NVARCHAR(20)    NULL,
    ContrasenaHash       NVARCHAR(256)   NULL,               -- NULL si aún no crea cuenta web
    FotoUrl              NVARCHAR(500)   NULL,
    Estado               NVARCHAR(20)    NOT NULL DEFAULT (N'Activo'),
    FechaRegistro        DATETIME2       NOT NULL DEFAULT (SYSDATETIME()),
    CONSTRAINT PK_Cliente PRIMARY KEY (ClienteId),
    CONSTRAINT UQ_Cliente_DNI UNIQUE (DNI),
    CONSTRAINT UQ_Cliente_Correo UNIQUE (Correo),
    CONSTRAINT CK_Cliente_Estado CHECK (Estado IN (N'Activo', N'Inactivo'))
    -- Nota: las etiquetas "Frecuente" / "Nuevo" del mockup de Clientes son
    -- estados DERIVADOS (según N° de reservas), no se almacenan aquí; se
    -- calculan con una vista o consulta (ver notas al final).
);
GO

/* ============================================================================
   6. RESERVAS (núcleo transaccional del sistema)
   ============================================================================ */

CREATE TABLE dbo.Reserva (
    ReservaId               INT             IDENTITY(1,1)   NOT NULL,
    CodigoReserva            NVARCHAR(12)    NOT NULL,           -- formato CA-XXXXX
    ClienteId                INT             NOT NULL,
    HabitacionId             INT             NOT NULL,
    CreadoPorUsuarioId       INT             NULL,               -- si el staff la registró
    FechaCheckIn             DATE            NOT NULL,
    FechaCheckOut            DATE            NOT NULL,
    NumAdultos                TINYINT         NOT NULL DEFAULT (1),
    NumNinos                  TINYINT         NOT NULL DEFAULT (0),
    Estado                    NVARCHAR(20)    NOT NULL DEFAULT (N'Pendiente'),
    Canal                     NVARCHAR(20)    NOT NULL DEFAULT (N'Directo'),
    PrecioTotal                DECIMAL(10,2)   NOT NULL,
    FechaCreacion              DATETIME2       NOT NULL DEFAULT (SYSDATETIME()),
    CONSTRAINT PK_Reserva PRIMARY KEY (ReservaId),
    CONSTRAINT UQ_Reserva_Codigo UNIQUE (CodigoReserva),
    CONSTRAINT FK_Reserva_Cliente FOREIGN KEY (ClienteId)
        REFERENCES dbo.Cliente (ClienteId),
    CONSTRAINT FK_Reserva_Habitacion FOREIGN KEY (HabitacionId)
        REFERENCES dbo.Habitacion (HabitacionId),
    CONSTRAINT FK_Reserva_Usuario FOREIGN KEY (CreadoPorUsuarioId)
        REFERENCES dbo.Usuario (UsuarioId),
    CONSTRAINT CK_Reserva_Estado CHECK (Estado IN
        (N'Pendiente', N'Confirmada', N'Bloqueada', N'Completada', N'Cancelada')),
    CONSTRAINT CK_Reserva_Canal CHECK (Canal IN (N'Directo', N'OTA')),
    CONSTRAINT CK_Reserva_Fechas CHECK (FechaCheckOut > FechaCheckIn),
    CONSTRAINT CK_Reserva_PrecioTotal CHECK (PrecioTotal >= 0)
);
GO

-- Servicios adicionales contratados dentro de una reserva (venta cruzada:
-- WiFi, AC, Jacuzzi, traslados... mencionada en Oportunidad del Business Case)
CREATE TABLE dbo.ReservaServicio (
    ReservaId           INT             NOT NULL,
    ServicioId           INT             NOT NULL,
    Cantidad              INT             NOT NULL DEFAULT (1),
    PrecioUnitario        DECIMAL(10,2)   NOT NULL,
    CONSTRAINT PK_ReservaServicio PRIMARY KEY (ReservaId, ServicioId),
    CONSTRAINT FK_ReservaServicio_Reserva FOREIGN KEY (ReservaId)
        REFERENCES dbo.Reserva (ReservaId) ON DELETE CASCADE,
    CONSTRAINT FK_ReservaServicio_Servicio FOREIGN KEY (ServicioId)
        REFERENCES dbo.Servicio (ServicioId),
    CONSTRAINT CK_ReservaServicio_Cantidad CHECK (Cantidad > 0)
);
GO

-- Comprobante / voucher de la reserva (mockup "Mis Reservas": botón Comprobante)
CREATE TABLE dbo.Comprobante (
    ComprobanteId        INT             IDENTITY(1,1)   NOT NULL,
    ReservaId              INT             NOT NULL,
    NumeroComprobante      NVARCHAR(30)    NOT NULL,
    FechaEmision            DATETIME2       NOT NULL DEFAULT (SYSDATETIME()),
    MontoTotal              DECIMAL(10,2)   NOT NULL,
    UrlPdf                  NVARCHAR(500)   NULL,
    CONSTRAINT PK_Comprobante PRIMARY KEY (ComprobanteId),
    CONSTRAINT UQ_Comprobante_Reserva UNIQUE (ReservaId),      -- 1 a 1 con Reserva
    CONSTRAINT UQ_Comprobante_Numero UNIQUE (NumeroComprobante),
    CONSTRAINT FK_Comprobante_Reserva FOREIGN KEY (ReservaId)
        REFERENCES dbo.Reserva (ReservaId)
);
GO

/* ============================================================================
   7. ASISTENTE VIRTUAL / PLANIFICADOR INTELIGENTE (log para BI y OE2)
   ============================================================================ */

-- Registra cada interacción del chatbot (presupuesto + clima -> sede
-- recomendada) para medir OE2 ("tasa de interacción exitosa") y el
-- indicador de BI "conversiones del asistente AI"
CREATE TABLE dbo.ConsultaChatbot (
    ConsultaId              INT             IDENTITY(1,1)   NOT NULL,
    ClienteId                 INT             NULL,               -- NULL si es visitante anónimo
    Presupuesto                NVARCHAR(20)    NOT NULL,
    Clima                      NVARCHAR(20)    NOT NULL,
    SedeRecomendadaId          INT             NULL,
    ReservaGeneradaId          INT             NULL,               -- si la consulta terminó en reserva
    FechaConsulta               DATETIME2       NOT NULL DEFAULT (SYSDATETIME()),
    CONSTRAINT PK_ConsultaChatbot PRIMARY KEY (ConsultaId),
    CONSTRAINT FK_Consulta_Cliente FOREIGN KEY (ClienteId)
        REFERENCES dbo.Cliente (ClienteId),
    CONSTRAINT FK_Consulta_Sede FOREIGN KEY (SedeRecomendadaId)
        REFERENCES dbo.Sede (SedeId),
    CONSTRAINT FK_Consulta_Reserva FOREIGN KEY (ReservaGeneradaId)
        REFERENCES dbo.Reserva (ReservaId),
    CONSTRAINT CK_Consulta_Presupuesto CHECK (Presupuesto IN
        (N'Económico', N'Estándar', N'Lujo')),
    CONSTRAINT CK_Consulta_Clima CHECK (Clima IN
        (N'Cálido', N'Frío', N'Templado'))
);
GO

CREATE INDEX IX_Reserva_Estado ON dbo.Reserva (Estado);
CREATE INDEX IX_Usuario_Sede ON dbo.Usuario (SedeId);
CREATE INDEX IX_ConsultaChatbot_Sede ON dbo.ConsultaChatbot (SedeRecomendadaId);
GO
